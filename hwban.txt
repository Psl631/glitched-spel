function runDiagnostics() {
            const fill = document.getElementById('loading-bar-fill');
            const txt = document.getElementById('diag-status-text');

            // Steg 1: RAM-minne (25% klart)
            setTimeout(() => {
                if (fill) fill.style.width = "25%";
                if (txt) txt.innerText = "CHECKING PHYSICAL MEMORY (RAM)... OK";
                
                // Steg 2: CPU-kärnor (50% klart)
                setTimeout(() => {
                    if (fill) fill.style.width = "50%";
                    if (txt) txt.innerText = "VERIFYING CPU ARCHITECTURE CORES... OK";
                    
                    // Steg 3: OS & Säkerhet (75% klart)
                    setTimeout(() => {
                        if (fill) fill.style.width = "75%";
                        if (txt) txt.innerText = "SECURING OS ARCHITECTURE TUNNELS... OK";
                        
                        // Steg 4: WebGL & GPU-kontroll (100% klart)
                        setTimeout(() => {
                            if (fill) fill.style.width = "100%";
                            if (txt) txt.innerText = "INITIALIZING 3D WEBGL GRAPHICS CORE... READY";

                            
                            // Kör nätverkskontrollen direkt när filen laddas i webbläsaren
                            verifyElmiaNetwork();
                            
                            // Kör hårdvarubevakningen och rensa laddskärmen efter en kort fördröjning
                            setTimeout(() => {
                                let canvas = document.createElement('canvas');
                                let gl = canvas.getContext('webgl') || canvas.getContext('experimental-webgl');
                                
                                if (!gl) {
                                    triggerHardwareBanScreen("no_webgl");
                                    return;
                                }

                                let gpuString = "";
                                let debugInfo = gl.getExtension('WEBGL_debug_renderer_info');
                                if (debugInfo) {
                                    gpuString = gl.getParameter(debugInfo.UNMASKED_RENDERER_WEBGL).toLowerCase();
                                }

                                let ua = navigator.userAgent.toLowerCase();
                                let hardwareReport = (ua + " " + gpuString);
                                let banKey = null;

                                if (navigator.deviceMemory && navigator.deviceMemory < 4) { banKey = "low_ram"; }
                                if (!banKey && (gpuString === "" || gpuString.includes("software renderer") || gpuString.includes("swiftshader"))) { banKey = "no_gpu"; }
                                if (!banKey) {
                                    if (hardwareReport.includes("celeron")) banKey = "celeron";
                                    else if (hardwareReport.includes("pentium")) banKey = "pentium";
                                    else if (hardwareReport.includes("atom")) banKey = "atom";
                                }
                                if (!banKey && navigator.hardwareConcurrency) {
                                    if (navigator.hardwareConcurrency === 1) banKey = "single_core";
                                    else if (navigator.hardwareConcurrency < 4) banKey = "slow_cores";
                                }
                                if (!banKey) {
                                    const botKeywords = ['googlebot', 'google-x', 'bingbot', 'yandexbot', 'ahrefsbot', 'spider', 'crawler', 'scrap', 'headless'];
                                    let isBotAgent = botKeywords.some(keyword => ua.includes(keyword)) || navigator.webdriver;
                                    if (isBotAgent) banKey = "bot";
                                }
                                if (!banKey) {
                                    if (hardwareReport.includes("playstation") || hardwareReport.includes("nintendo") || hardwareReport.includes("wii") || hardwareReport.includes("xbox")) banKey = "console";
                                    else if (hardwareReport.includes("smart-tv") || hardwareReport.includes("smarttv") || hardwareReport.includes("googletv") || hardwareReport.includes("appletv") || hardwareReport.includes("tizen") || hardwareReport.includes("webos") || ua.includes("lg netcast") || ua.includes("vizio")) banKey = "tv";
                                    else if (hardwareReport.includes("quest ") || hardwareReport.includes("oculus") || hardwareReport.includes("vive") || hardwareReport.includes("pico") || hardwareReport.includes("openvr") || ua.includes("vision pro") || ua.includes(" xr ") || ua.includes("spatial")) banKey = "vr";
                                    else if (hardwareReport.includes("tesla") || hardwareReport.includes("carplay") || hardwareReport.includes("automotive") || ua.includes("polestar") || ua.includes("volvo_car") || ua.includes("infotainment")) banKey = "car";
                                    else if (hardwareReport.includes("watch") || hardwareReport.includes("wear os") || ua.includes("wearable") || ua.includes("applewatch") || ua.includes("fitbit")) banKey = "smartwatch";
                                    else if (hardwareReport.includes("cros") || hardwareReport.includes("chromebook")) banKey = "chromebook";
                                    else if (hardwareReport.includes("kindle") || hardwareReport.includes("nook") || hardwareReport.includes("bookeen") || ua.includes("kobo") || ua.includes("ereader")) banKey = "ereader";
                                    else if (hardwareReport.includes("alexa") || hardwareReport.includes("hub") || hardwareReport.includes("smart-home") || ua.includes("smarthome") || ua.includes("refrigerator") || ua.includes("washer") || ua.includes("dryer") || ua.includes("printer") || ua.includes("smart-oven")) banKey = "smarthome";
                                    else if (hardwareReport.includes("steam deck") || hardwareReport.includes("steamos")) banKey = "steamdeck";
                                    else if (ua.includes("anbernic") || ua.includes("miyoo") || ua.includes("powkiddy") || ua.includes("retroid pocket")) banKey = "retro_handheld";
                                    else if (ua.includes("point of sale") || ua.includes("pos-terminal") || ua.includes("izettle") || ua.includes("clover-device") || ua.includes("kiosk-mode")) banKey = "pos_terminal";
                                    else if (hardwareReport.includes("linux") && !hardwareReport.includes("android")) banKey = "linux";
                                    else if (/android|iphone|ipad|ipod|iemobile|blackberry|kindle/i.test(ua)) banKey = "mobile";
                                }
                                if (!banKey) {
                                    if (ua.includes("windows")) { if (!ua.includes("windows nt 10.0")) banKey = "old_windows"; }
                                    else if (ua.includes("macintosh")) {
                                        let match = ua.match(/mac os x (\d+)[_.](\d+)/);
                                        if (match) { let majorVersion = parseInt(match[1]); if (majorVersion < 15 && !ua.includes("version/15") && !ua.includes("version/16")) banKey = "old_mac"; }
                                    } else if (!ua.includes("mac") && !ua.includes("win")) { banKey = "invalid_os"; }
                                }

                                if (banKey) { 
                                    triggerHardwareBanScreen(banKey);
                                } else { 
                                    document.getElementById('diagnostic-screen').style.display = 'none';
                                    document.getElementById('start-screen').style.display = 'block';
                                    updateLeaderboardUI();
                                }
                            }, 600); // Andrum efter 100% klart innan menyn laddas
                        }, 400);
                    }, 400);
                }, 400);
            }, 400);
        }
