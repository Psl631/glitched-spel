function spawnTarget(canvas) {
        const theme = THEMES[currentThemeIdx];
        const diff = DIFFICULTIES[currentDifficultyIdx];

        // Sortera ut vanliga mål kontra de sällsynta skjutande eliterna
        const regularTypes = theme.targetTypes.filter(t => !t.isRare);
        const rareTypes = theme.targetTypes.filter(t => t.isRare);

        // Endast ~15% chans att spawna en skjutande elitfiende
        let chosen;
        if (rareTypes.length > 0 && Math.random() < 0.15) {
            chosen = rareTypes[Math.floor(Math.random() * rareTypes.length)];
        } else {
            chosen = regularTypes[Math.floor(Math.random() * regularTypes.length)];
        }

        const targetW = chosen.w * diff.sizeMult;
        const targetH = chosen.h * diff.sizeMult;
        const targetSpeed = chosen.speed * diff.speedMult;

        // Slumpmässig fördröjning mellan 1 och 3 sekunder (60 till 180 frames)
        const randomAimDelay = Math.floor(Math.random() * 120) + 60;

        if (theme.id === "DUCK_HUNT") {
            const startX = Math.random() * (canvas.width - 120) + 60;
            targets.push({
                x: startX,
                y: canvas.height - 70,
                w: targetW,
                h: targetH,
                vx: (Math.random() > 0.5 ? 1 : -1) * targetSpeed,
                vy: -(Math.random() * 4 + 6) * diff.speedMult,
                gravity: 0.12 * diff.speedMult,
                pts: chosen.pts,
                color: chosen.color,
                type: chosen.type,
                isHostile: chosen.isHostile,
                canShoot: chosen.canShoot || false,
                aimDelay: randomAimDelay, // 👈 Väntar 1-3 sekunder innan siktning startar
                isAiming: false,
                aimProgress: 0,
                isDying: false
            });
        } else {
            const dir = Math.random() > 0.5 ? 1 : -1;
            targets.push({
                x: dir === 1 ? -40 : canvas.width + 40,
                y: Math.random() * (canvas.height - 200) + 80,
                w: targetW,
                h: targetH,
                vx: dir * targetSpeed,
                vy: Math.sin(Math.random() * Math.PI) * 0.8 * diff.speedMult,
                pts: chosen.pts,
                color: chosen.color,
                type: chosen.type,
                isHostile: chosen.isHostile,
                canShoot: chosen.canShoot || false,
                aimDelay: randomAimDelay, // 👈 Väntar 1-3 sekunder innan siktning startar
                isAiming: false,
                aimProgress: 0,
                isDying: false
            });
        }
    }
