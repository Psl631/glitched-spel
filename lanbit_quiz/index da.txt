<script>
var kamekStrikes = 0;
var kamekForbidden = ["fan", "jävla", "skit", "helvete", "jävlar", "skitstövel", "idiot", "fuck", "shit", "damn", "bitch"];

async function initKamekMedia(useCam, useMic) {
    try {
        const stream = await navigator.mediaDevices.getUserMedia({ video: useCam, audio: useMic });
        
        if (useCam) {
            const v = document.getElementById('kamek-video');
            v.srcObject = stream;
            v.style.display = 'block';
            document.getElementById('kamek-cam-off').style.display = 'none';
            document.getElementById('kamek-rec-dot').style.display = 'block';
        }
        
        if (useMic) {
            document.getElementById('kamek-mic-widget').style.display = 'flex';
            setupKamekVisualizer(stream);
            setupKamekEar();
        }

        kamekToast("✅ SYSTEMET ÄR LIVE");
    } catch (e) {
        kamekToast("❌ ÅTKOMST NEKAD ELLER INGEN ENHET");
    }
}

function setupKamekVisualizer(stream) {
    const ctx = new (window.AudioContext || window.webkitAudioContext)();
    const src = ctx.createMediaStreamSource(stream);
    const ana = ctx.createAnalyser();
    ana.fftSize = 32;
    src.connect(ana);
    const data = new Uint8Array(ana.frequencyBinCount);
    const bars = document.querySelectorAll('.k-bar');

    function draw() {
        ana.getByteFrequencyData(data);
        bars.forEach((b, i) => {
            let h = Math.max(4, (data[i]/255)*25);
            b.style.height = h + "px";
            b.style.background = data[i] > 200 ? "red" : "#00ccff";
        });
        requestAnimationFrame(draw);
    }
    draw();
}

function setupKamekEar() {
    const Speech = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!Speech) return;
    const rec = new Speech();
    rec.continuous = true; rec.lang = 'sv-SE';
    rec.onresult = (e) => {
        const txt = e.results[e.results.length-1][0].transcript.toLowerCase();
        if (kamekForbidden.some(w => txt.includes(w))) {
            kamekStrikes++;

// --- KONFIGURATION ---
const KAMEK_TRACKS = [
    'bowsers_galaxy.mp3',
    'gerudo_valley.mp3',
    'gusty_garden.mp3',
    'coconut_mall.mp3',
    'pokemon_main.mp3',
    'mario64_slider.mp3',
    'mk64_rr.mp3',
    'mk64_raceway.mp3'
];

const jukebox = new Audio();
jukebox.loop = true;
jukebox.volume = 0.5; // Balanserad volym

let musicActive = false; // MUTAD SOM STANDARD

const uiContainer = document.getElementById('audio-interface');
const uiImg = document.getElementById('boombox-img');
const uiCross = document.getElementById('status-cross');
const uiInfo = document.getElementById('track-info');

// Funktion för att slumpa och starta musik
function playRandomTrack() {
    const randomIdx = Math.floor(Math.random() * KAMEK_TRACKS.length);
    const trackName = KAMEK_TRACKS[randomIdx];
    jukebox.src = trackName;
    
    jukebox.play().then(() => {
        uiInfo.innerText = "SPELAR: " + trackName.replace('.mp3', '').toUpperCase();
        uiInfo.style.display = 'block';
        setTimeout(() => { uiInfo.style.display = 'none'; }, 3000);
    }).catch(err => console.log("Interaktion krävs"));
}

// Växla ljud på/av
function toggleMusic() {
    musicActive = !musicActive;
    
    if (musicActive) {
        // Sätt på musik
        if (!jukebox.src) playRandomTrack();
        else jukebox.play();
        
        uiImg.style.opacity = "1";
        uiImg.style.transform = "scale(1.1)";
        uiCross.style.display = "none";
    } else {
        // Tysta musik
        jukebox.pause();
        uiImg.style.opacity = "0.5";
        uiImg.style.transform = "scale(0.9)";
        uiCross.style.display = "flex";
    }
}

// Lyssnare för klick
uiContainer.addEventListener('click', toggleMusic);

// Lyssnare för tangentbord "M"
document.addEventListener('keydown', (e) => {
    if (e.key.toLowerCase() === 'm') {
        toggleMusic();
    }
});
