    function generateQuestion() {
    let diff = Math.floor(score / 2000);
    // Operatörer låses upp gradvis baserat på poäng för att öka kaoset
    let ops = ['+', '-', '*'];
    if (score > 2000) ops.push('/');
    if (score > 4000) ops.push('%');
    if (score > 6000) ops.push('^', 'sqrt');

    const op = ops[Math.floor(Math.random() * ops.length)];
    let n1, n2;

    if (op === '/') {
        // Division: Garanterar heltal genom att multiplicera baklänges
        n2 = Math.floor(Math.random() * 9) + 2;
        currentAnswer = Math.floor(Math.random() * 10) + 1;
        n1 = currentAnswer * n2;
        document.getElementById('question').innerText = n1 + " / " + n2;
    } 
    else if (op === '%') {
        // Procent: Använder jämna procentsatser för huvudräkning
        let percentages = [10, 20, 25, 50, 75];
        n2 = percentages[Math.floor(Math.random() * percentages.length)];
        n1 = (Math.floor(Math.random() * 10) + 1) * 20;
        currentAnswer = (n2 / 100) * n1;
        document.getElementById('question').innerText = n2 + "% AV " + n1;
    }
    else if (op === '^') {
        // Potens: Upphöjt till 2 (Kvadrat)
        n1 = Math.floor(Math.random() * 10) + 2;
        currentAnswer = Math.pow(n1, 2);
        document.getElementById('question').innerText = n1 + "²";
    }
    else if (op === 'sqrt') {
        // Roten ur: Garanterar jämna rötter
        currentAnswer = Math.floor(Math.random() * 11) + 2;
        n1 = currentAnswer * currentAnswer;
        document.getElementById('question').innerText = "√" + n1;
    }
    else if (op === '*') {
        n1 = Math.floor(Math.random() * (10 + diff)) + 2;
        n2 = Math.floor(Math.random() * (10 + diff)) + 2;
        currentAnswer = n1 * n2;
        document.getElementById('question').innerText = n1 + " × " + n2;
    }
    else if (op === '-') {
        n1 = Math.floor(Math.random() * (50 + diff)) + 20;
        n2 = Math.floor(Math.random() * 19) + 2;
        currentAnswer = n1 - n2;
        document.getElementById('question').innerText = n1 + " - " + n2;
    }
    else {
        // Standard: Addition
        n1 = Math.floor(Math.random() * (50 + diff)) + 2;
        n2 = Math.floor(Math.random() * (50 + diff)) + 2;
        currentAnswer = n1 + n2;
        document.getElementById('question').innerText = n1 + " + " + n2;
    } // <--- Slut på if/else-kedjan

    setupOptions(currentAnswer);
} // <--- VIKTIGT: Denna stänger hela generateQuestion-funktionen

    function setupOptions(correct) {
        let opts = [correct];
        while(opts.length < 4) {
            let a = correct + (Math.floor(Math.random()*16)-8);
            if(a > 0 && !opts.includes(a)) opts.push(a);
        }
        opts.sort(() => Math.random()-0.5);
        const grid = document.getElementById('answer-options');
        grid.innerHTML = '';
        opts.forEach(o => {
            const b = document.createElement('button'); b.className = 'answer-btn'; b.innerText = o;
            b.onclick = () => {
                if (o === correct) { 
                    score += 150 + (Math.floor(score / 1000) * 10); 
                    timeLeft += 2;
                    new Audio('pling.mp3').play().catch(()=>{});
                    showToast("TUR ELLER SKICKLIGHET?", "#00ff00");
                } else {

                    setTimeout(() => document.body.style.backgroundColor = '', 300);
                    timeLeft -= 5;
                }
                document.getElementById('display-score').innerText = score;
                generateQuestion(); 
            };
            grid.appendChild(b);
        });
    }

    function triggerAlarm() {
        alarmActive = true;
        const overlay = document.getElementById('alarm-overlay'), taskArea = document.getElementById('alarm-task-area'), mainBtn = document.getElementById('main-snooze');
        overlay.style.display = 'flex'; taskArea.innerHTML = ''; mainBtn.style.display = 'none';
        
        let modes = ["SIMPLE", "DOT", "WORD", "MULTI_CLICK", "HOLD_KEY", "Q_SPAM"];
        snoozeMode = modes[Math.floor(Math.random() * modes.length)];

        if (snoozeMode === "SIMPLE") {
            document.getElementById('alarm-msg').innerText = "VAKNA!!!";
            mainBtn.style.display = 'block';
        } else if (snoozeMode === "MULTI_CLICK") {
            let clicks = 6 + Math.floor(score/5000);
            document.getElementById('alarm-msg').innerText = "KLICKA " + clicks + " GÅNGER!";
            mainBtn.style.display = 'block';
            mainBtn.onclick = () => { clicks--; document.getElementById('alarm-msg').innerText = "KLICKA " + clicks + " GÅNGER!"; if(clicks<=0) snooze(); };
        } else if (snoozeMode === "DOT") {
            let dotsToFind = 1 + Math.floor(score/8000); // Ökar mer sällan nu
            document.getElementById('alarm-msg').innerText = "JAGA PRICKEN (" + dotsToFind + ")!";
            for (let i = 0; i < dotsToFind; i++) {
                const dot = document.createElement('div');
                // Gjorde pricken 60px istället för 50px för att den ska vara lättare att träffa
                dot.style.cssText = "width:60px; height:60px; background:yellow; border-radius:50%; position:absolute; cursor:pointer; transition:0.6s; z-index:2000; border: 3px solid black;";
                taskArea.appendChild(dot);
                const move = () => { 
                    dot.style.top = (Math.random()*70+15) + "%"; 
                    dot.style.left = (Math.random()*70+15) + "%"; 
                };
                move();
                // Sänkt hastighet: 800ms starttid istället för 400ms
                let moveInt = setInterval(move, 800 - Math.min(400, (score / 100))); 
                dot.onclick = () => { 
                    dot.remove(); 
                    clearInterval(moveInt); 
                    dotsToFind--; 
                    if(dotsToFind <= 0) snooze(); 
                }; }
            } else if (snoozeMode === "WORD") {
            const words = ["KAMEK", "LANBIT", "KUNGSBACKA", "DATORMUS", "HÅRDDISKFRAGMENTERING", "TANGENTBORD", "DATORMUS", "ADRESS"];
            const target = words[Math.floor(Math.random() * words.length)];
            document.getElementById('alarm-msg').innerHTML = "STAVA: <span style='color:yellow'>" + target + "</span>";
            const input = document.createElement('input'); input.id = "snooze-input";
            input.oninput = () => { if(input.value.toUpperCase() === target) snooze(); else if(!target.startsWith(input.value.toUpperCase())) { new Audio('boooop.mp3').play(); endGame("STAVFEL!"); } };
            taskArea.appendChild(input); setTimeout(() => input.focus(), 50);
        } else if (snoozeMode === "Q_SPAM") {
            qSpamCount = 0;
            document.getElementById('alarm-msg').innerText = "SPAMMA Q 10 GÅNGER! (5s)";
            dotTimeout = setTimeout(() => { if(alarmActive) endGame("FÖR SEGT SPAM!"); }, 5000);
        } else if (snoozeMode === "HOLD_KEY") {
            const keys = ['A', 'S', 'D', 'F'];
            targetHoldKey = keys[Math.floor(Math.random() * keys.length)];
            document.getElementById('alarm-msg').innerText = `HÅLL IN '${targetHoldKey}' I 2 SEK!`;
            holdStartTime = 0;
            alarmTaskInterval = setInterval(() => { if(holdStartTime > 0 && Date.now() - holdStartTime >= 2000) snooze(); }, 100);
        }
// Lista på slumpade alarmljud
        const alarmSounds = ['alarm_buzzer.mp3', 'clock_alarm.mp3', 'cukoo_clock.mp3', 'classic_alarm.mp3'];

    function snooze() {
        alarmActive = false; document.getElementById('alarm-overlay').style.display = 'none';
        if(currentAlarmAudio) { currentAlarmAudio.pause(); currentAlarmAudio.currentTime = 0; }
        clearInterval(moveInterval); clearInterval(alarmTaskInterval); clearTimeout(dotTimeout);
        holdStartTime = 0; snoozeMode = "";
        showToast("TUR FÖR DIG, DIN SOPA...", "#ffcc00");
    }
